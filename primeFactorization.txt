Prime Factorization

import java.util.*;
public class Main
{
	public static void main(String[] args) {
        
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter your no");
		int n = scn.nextInt();
		System.out.println("Your no: " + n);
		
		for(int div = 2; div*div <= n; div++){
		    while(n % div == 0){
		        n = n / div;
		        System.out.println(div);
		    }
		}
		
		if(n != 1){
		    System.out.println(n);
		}

	}
}
