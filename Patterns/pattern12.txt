import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter the no of rows: ");
		int n = scn.nextInt();
		System.out.println("no of rows: " + n);
		
		System.out.println("Pattern: ");
		
		int a = 0;
		int b = 1;
		
		for(int i = 1; i <= n; i++){
		    
		    for(int j = 1; j <= i; j++){
		        System.out.print(a + "\t");
		        int c = a + b;
		        a = b;
		        b = c;
		    }
		    
		    System.out.println();
		}
	}
}
