import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter the no of rows: ");
		int n = scn.nextInt();
		System.out.println("no of rows: " + n);
		
        System.out.println("Pattern: ");
		
		int sp = 2 * n - 3;
		int st = 1;
		
		for(int i = 1; i <= n; i++){
		    int val = 1;
		    
		    for(int j = 1; j <= st; j++){
		        System.out.print(val + "\t");
		        val++;
		    }
		    
		    
		    for(int j = 1; j <= sp; j++){
		        System.out.print("\t");
		    }
		    
		    if(i == n){
		        st--;
		        val--;
		    }
		    
		    for(int j = 1; j <= st; j++){
		        val--;
		        System.out.print(val + "\t");
		    }
		    
		    st++;
		    sp -= 2;
		    
		    System.out.println();
		}
	}
}
