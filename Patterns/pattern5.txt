import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter no: ");
		int n = scn.nextInt();
        System.out.println("no: " + n);
		
		int sp = n/2 ;
		int st = 1 ;
		
		for(int i = 1; i <= n; i++){
		  //  System.out.println(sp + "," + st);
		  
		    for(int j = 1; j <= sp; j++){
		        System.out.print("\t");
		    }
		    
		    for(int j = 1; j <= st; j++){
		        System.out.print("*\t");
		    }
		    
		    if(i <= n/2){
		        sp-- ;
		        st += 2 ;
		    } else {
		        sp++ ;
		        st-= 2 ;
		    }
		    
		    System.out.println();
		}
	}
}
