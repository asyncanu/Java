import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter the no for table: ");
		int x = scn.nextInt();
		System.out.println("no for table : " + x);
		
		System.out.println("Table: ");
		
		for(int i = 0; i <= 10; i++){
		    int val = x * i ;
		    System.out.println(x + " * " + i + " = " + val);
		}
	}
}
