import java.util.*;
public class Main
{
	public static void main(String[] args) {
    Scanner scn = new Scanner(System.in);
	    
	System.out.println("Enter your number:");
	int n = scn.nextInt();
	System.out.println("your number:" + n);
	
	int sp = 0;
	int star = n;
	
	for(int i = 1; i <= n; i++){
	    
	    for(int j = 1; j <= sp; j++){
	        //  System.out.println(sp + "," + star);
		      System.out.print("\t");
		  }
		  
		  for(int j = 1; j <= star; j++){
		      System.out.print("*\t");
		  }
		    sp++ ;
		    star-- ;
		    System.out.println();
	    }

    }
}

5
your number: 5

  * * * *
