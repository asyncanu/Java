import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter the no of rows: ");
		int n = scn.nextInt();
		System.out.println("No of rows: " + n);
		
		int sp = n/2;
		int st = 1;
		int val = 1;
		
		for(int i = 1; i <= n; i++){
		    for(int j = 1; j <= sp; j++){
		        System.out.print("\t");
		    }
		    
		    int cval = val;
		    for(int j = 1; j <= st; j++){
		        System.out.print(cval + "\t");
		        
		        if(j <= n/2){
		            cval++;
		        } else {
		            cval--;
		        }
		    }
		    
		    if(i <= n/2){
		        sp--;
		        st+= 2;
		        val++;
		    } else {
		        sp++;
		        st-= 2;
		        val--;
		    }
		    
		    
		    System.out.println();
		}
	}
}
