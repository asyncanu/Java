import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter your no: ");
		int n = scn.nextInt();
		System.out.println("no: " + n);
		
		System.out.println("Pattern: ");
		
		for(int i = 1; i <= n; i++){
		    
		    for(int j = 1; j <= n; j++){
		        
		        if(i + j == n + 1){
		            System.out.print("*\t");
		        } else{
		            System.out.print("\t");
		        }
		    }
		    
		    System.out.println();
		}
	}
}
