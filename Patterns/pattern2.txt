import java.util.*;
public class Main
{
	public static void main(String[] args) {
	    Scanner scn = new Scanner(System.in);
	    
		System.out.println("Enter no");
		int n = scn.nextInt();
		System.out.println("your no: " + n );
		
		for(int i = n; i >= 1; i--){
		    
		    for(int j = 1; j <= i; j++){
		        System.out.print("*\t");
		    }
		    
		    System.out.println(); 
		}
		
    }
}

5
